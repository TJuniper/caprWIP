# SC016 OEWsPalatalGlide — WS glide spelling of back vowels after initial ġ.
# Adjudication: Germanic/docs/sound_changes/audits/sc016-017-adjudication.md.
#
# After word-initial ġ /j/, Old English scribes wrote a palatal-glide letter
# before back vowels; the normalized (late) WS digraph is ⟨eo⟩:
#   - o-subcase (< NWGmc u-lowering): *jok → ġeoc 'yoke'
#     (Brunner §92.1b pp.64-65; Bülbring §299 p.120 "aus wg. *jok";
#      Campbell §115 p.43 lists geoc under u > o, cf. §§172-173)
#   - u-subcase (lowering blocked by following high vowel): *juguθ → ġeoguþ
#     (Brunner §92.1a; Bülbring §298; Campbell §172; Luick §169 pp.158-159)
# R/T p.5: "After word-initial /j/ followed by a back vowel that practice
# was universal … geoc 'yoke' is /jok/" — the ⟨e⟩ is orthographic; Hogg,
# CHEL I p.112 concurs (back-vowel cases "never anything more than an
# orthographic variation", contra Campbell §176). The older school
# (Campbell §44; Luick §169 "schwebende Diphthonge"; Brunner §92;
# Bülbring §§47, 298-299) reads the same spellings as rising diphthongs;
# either way the nucleus remains the back vowel and the practice is an
# OE-period development fed by NWGmc u-lowering (SC017 < SC016).
#
# Restricted to word-initial ġ: after /tʃ ʃ/ the practice was not universal
# (R/T p.5; Hogg p.112 on sc(e)op-type inconsistency). The ā/ō subcases
# (geara, geōmor) are documented in the adjudication but unexercised by
# the corpus. Early-WS io (gioc) and Anglian iu (giung) spellings are
# dialectal variants not modeled here.
define OEWsPalatalGlide [
    {*ó} -> {*éo} || .#. ġ _ ,
    {*ú} -> {*éo} || .#. ġ _ ,
    {*o} -> {*eo} || .#. ġ _ ,
    {*u} -> {*eo} || .#. ġ _
];

# environment (short V + single C + back V). They do NOT trigger heavy-syllable
# apocope. Short diphthongs from BREAKING always occur in a 2-C environment,
# which is handled by the independent "short V + 2 C" clause.

# breaking products (which required a cluster environment) count as HEAVY.

# Rhotacism: surviving *z > *r (R/T vol.2 p.87: "first *z was lost in
# a variety of environments ..., then all surviving *z became *r
# across the whole WGmc area"). Executable position matches that
# chronology: this rule is composed inside EnglishProtoToOE
# immediately AFTER all three final-*z losses (RootNounNomZLoss,
# EAFFinalZDeletion, MonosyllabicFinalZLoss), so word-final *-z is
# gone before rhotacism applies — deletion is *z-sensitive and so
# must precede the *z/*r merger (Crist 2002 §6). Right-context `?`
# requires *some* symbol to follow *z, permitting every medial
# position (VzV, VzC) while blocking the word boundary; a stricter
# V_V context was tried and rejected (regresses medial VzC forms
# such as *mízdō). See DEV_NOTES §17.10.24.
define EAFRhotacism [
    {*z} -> {*r} || EnglishStarVocalic _ ?
];

# PGmc *gm → *m: spirant loss before *m (Kroonen p.511, p.101)

# the *g is voiced [ɣ], not voiceless *x. The voicing comes from Verner's Law

# PGmc final *-z deletion. R/T vol.2 §3.3.1 (p.98): z-loss preceded
# rhotacism — final *-z was never rhotacized.
#
# As of DEV_NOTES §17.10.23, z-loss is PLAIN: the earlier bundled rule
# PGmcFinalOZShortening (which also collapsed *-ōz → *-æ, Option γ) was
# removed in favour of R/T's own chronology. The z-loss-exposed *-ō is
# handled later by PWGmcSurvivingBimoricOUnrounding (R/T §3.1 pp.58-59),
# which matches R/T's "surviving bimoric *-ō → PWGmc *-a" (represented
# here as long *-ā so length can be tracked through AFB until R/T's own
# OE-stage shortening at §6.8.3). The full staged chain is:
#
#   *-ōz (PGmc)
#    ↓ EAFFinalZDeletion              [this rule, step 3]
#   *-ō  (PWGmc, z-lost)
#    ↓ PWGmcSurvivingBimoricOUnrounding [step 5, word-final]
#   *-ā  (PWGmc/WGmc)
#    ↓ EAFBrightening          [step 6, word-final long-vowel clause]
#   *-ǣ  (pre-OE)
#    ↓ OEUnstressedLongVowelShortening6 [step 7, existing rule]
#   *-æ  (eOE)
#    ↓ OEUnstressedAEMerger             [step 8]
#   -e   (OE)
#
# Ordering constraint: z-deletion runs BEFORE EAFRhotacism (which is
# composed after MonosyllabicFinalZLoss inside EnglishProtoToOE) so
# that word-final *-z is lost, not rhotacized.
# The PNWGmc *-ō → *-u raising (PNWGmcFinalLongORaising, R/T pp.30-31)
# must run BEFORE this rule so that pre-existing word-final *-ō
# (nom.sg. *gebō etc.) raises while *-ōz (sheltered by *-z) does not.
# That ordering is enforced in OldEnglishCore (see below).
#
# THREE-RULE DECOMPOSITION (2026, SC020 audit): the former unconditional
# body ({*z} -> 0 || _ .#.) conflated three historically distinct
# developments. It is now split into:
#   SC096 RootNounNomZLoss        — consonant-stem nominative *-z loss
#                                   (pre-PWGmc; monosyllabic C+z proxy)
#   SC020 EAFFinalZDeletion       — PWGmc loss of final *-z in unstressed
#                                   syllables (this rule, narrowed to
#                                   polysyllables; R/T §3.1.1 pp.44-45)
#   SC097 MonosyllabicFinalZLoss  — later northern WGmc loss of *-z in
#                                   stressed monosyllables, with
#                                   compensatory lengthening (R/T §3.3.1
#                                   pp.86-87; corpus witness: who
#                                   *xwáz > hwā)
# The three environments partition all word-final *z inputs exactly; see
# Germanic/docs/sound_changes/audits/sc020-three-rule-adjudication.md and
# Dossiers A/B/C in the same directory. Identifier names are provisional
# (global rename pass deferred).
#
# SC020 environment: word-final *z in an unstressed syllable. Two
# implementable clauses:
#   (i)  POLYSYLLABLE — two vowel nuclei separated by at least one
#        consonant somewhere before the final *z (every polysyllabic
#        final syllable in the corpus is post-stress);
#   (ii) post-consonantal *z in a monosyllable. By this cascade point
#        SC096 RootNounNomZLoss (composed at the head of
#        EnglishProtoToOE, before PWGmcIjContraction) has already
#        consumed the genuine root-noun nominatives, so the only forms
#        reaching clause (ii) are monosyllables CREATED by contraction
#        from historically polysyllabic inputs — in the corpus exactly
#        friend *fríjōndz → *fríundz, whose final syllable was
#        unstressed when this change applied (adjudication memo §5:
#        friend/milk/month belong to SC020, per R/T 2014: 44 *fadurz).
define EAFFinalZDeletion Ctx([
    [{*z} -> 0 ||
    .#. ?* EnglishStarVocalic [EnglishStarConsonant | EnglishPalatalConsonant]+ EnglishStarVocalic ?* _ .#.,
    .#. [EnglishStarConsonant | EnglishPalatalConsonant]* EnglishStarVocalic+ [EnglishStarConsonant | EnglishPalatalConsonant]+ _ .#.]
]);
define EAFFinalZLoss EAFFinalZDeletion;

# SC098: PWGmc apocope of word-final short high vowels after heavy
# syllables in unstressed words (R/T 2014: 57-58: "Short high vowels were
# also lost after heavy syllables in unstressed words ... OE iow 'you
# (dat. pl.)' definitely does [show early apocope] (since it does not
# exhibit i-umlaut)"). An exceptionless, prosodically conditioned law:
# like Verner's law it is governed by accent — here the absence of word
# stress in clitic sentence forms. The PWGmc doublet *iuwi ~ *iuw (R/T
# 2014: 41-42) is regular sentence sandhi: the stressed sentence variant
# kept *-i, the unstressed one lost it, and OE ēow continues the
# unstressed variant, as its lack of i-umlaut proves. Distinct from
# SC006 third-syllable i-apocope: stressed disyllables keep final *-i
# and umlaut (R/T 2014: 55; *gastiz > giest, *fūri > fȳr untouched).
# Sentence context is separate SOURCE metadata, not lexical accent.
# *ᵘ selects weak-final context; *ᶜ selects weak-nonfinal context.
# Nonfinality survives as a right-boundary annotation until SC063 has
# applied; its cleanup does not change SC063's vowel-loss clauses.
# Fed by SC020 (z-loss makes the *-i final); bleeds SC055 i-umlaut. See
# docs/sound_changes/audits/sc098-dossier-unstressed-word-final-i-apocope.md.
define PWGmcUnstressedWordFinalIApocope [
    [ [{*i}|{*u}] -> 0 || .#. {*ᵘ} ?* [
        [EnglishStarLongVowel | EnglishStarLongDiphthong] EnglishStarConsonant*
        | EnglishStarShortVowel EnglishStarConsonant EnglishStarConsonant+
    ] _ .#. ]
    .o. [{*ᵘ} -> 0]
    .o. [
        [{*ᶜ}:0 [EnglishStarAlphabet - {*ᶜ}]* 0:{*ᶜ}]
        | [EnglishStarAlphabet - {*ᶜ}]*
    ]
];

# SC096: consonant-stem (root-noun) nominative *-z loss. Historically a
# MORPHOLOGICAL development: the nom.sg. marker *-z was eliminated after
# stem-final consonant in athematic consonant stems, complete before
# PWGmc (R/T 2014: 118, 306, 313; Bammesberger 1990: 190-93; Dossier A).
# The FST states it through the phonological proxy "word-final *z after
# a consonant in a monosyllable", which in this corpus is exactly
# coextensive with the morphological class (book *bōkz, flea *fláuxz,
# goose *gánsz, louse *lūsz — 4 firings) PROVIDED the rule is composed
# at the head of EnglishProtoToOE, BEFORE PWGmcIjContraction: the
# contraction turns historically polysyllabic *fríjōndz into
# monosyllabic *fríundz, which this rule must never see in contracted
# shape (friend lost *-z by SC020, not here — memo §5). GUARD: if a
# future corpus row presents a monosyllabic C+z form at the head of the
# pipeline that is NOT a consonant-stem nominative, the proxy breaks
# and this rule must be re-scoped. The firing population is pinned
# fail-closed to book/flea/goose/louse by
# tests/test_final_z_firing_populations.py: any fifth firing (or any
# new TSV protoform matching the proxy environment) fails the suite
# and forces historical adjudication.
define RootNounNomZLoss Ctx([
    [{*z} -> 0 || .#. [EnglishStarConsonant | EnglishPalatalConsonant]* EnglishStarVocalic+ [EnglishStarConsonant | EnglishPalatalConsonant]+ _ .#.]
]);

# SC097: northern WGmc loss of word-final *-z in stressed monosyllables,
# with compensatory lengthening of a short nucleus (R/T 2014: 86-87:
# *hwaz > hwā, *maiz > mā — mā only via later ai-monophthongization;
# R/T print cow's history with question marks, so kū is NOT used as a
# principal control. OHG rhotacizes instead — ir, wir, mēr). Post-PWGmc,
# northern WGmc/Ingvaeonic scope (Dossier C). ONE selected corpus
# witness: who *xwáz > hwā (corpus-maturation pass 01; the corpus
# otherwise selects oblique cells for cow/night/breast/meed). Also
# validated by synthetic unit controls (*hwaz > *hwā, *hiz > *hī with
# compensatory lengthening; *maiz > *mai with the bimoric nucleus
# preserved — see test_final_z_firing_populations.py). Bimoric nuclei
# (long vowels, diphthongs) lose *-z without further lengthening.
define MonosyllabicFinalZLoss [
    {*a} -> {*ā}, {*á} -> {*ā},
    {*e} -> {*ē}, {*é} -> {*ḗ},
    {*i} -> {*ī}, {*í} -> {*ḯ},
    {*o} -> {*ō}, {*ó} -> {*ō},
    {*u} -> {*ū}, {*ú} -> {*ū}
        || .#. [EnglishStarConsonant | EnglishPalatalConsonant]* _ {*z} .#.
] .o. [
    {*z} -> 0 || .#. [EnglishStarConsonant | EnglishPalatalConsonant]* EnglishStarVocalic+ _ .#.
];

# PWGmc Early i-Apocope (Brunner §145, R/T vol.2 p.141)
# Final *-i lost in 3rd/4th syllables from stress, BEFORE i-umlaut.
# Brunner §145: "Ursprüngliches -i war gemeingerm. abgefallen... nach unbetonten 
# Silben (also in dritten und vierten Silben von der betonten aus gezählt)."
# Brunner §145 Anm.1: "Der frühe gemeingerm. Abfall dieser -i ist dadurch erwiesen,
# daß sie ae. keinen i-Umlaut mehr bewirkten."
# 
# Key evidence: forms like *júgunθiz → ġeoguþ show NO i-umlaut of medial *u,
# proving the final *-i was already lost before i-umlaut applied.
# R/T p.141: "*jugunþi > *juguþ > OE geoguþ ~ iuguþ"
#
# Pattern: stressed-V + C+ + V + C+ + final *i [+ optional *z] → delete the *i
# This targets the 3rd syllable position (σ́.σ.σi[z]).
#
# The optional *-z right-context handles dat./inst.pl. *-amiz / *-umiz
# (and parallel *-imiz / *-umiz / *-ōmiz across stem classes), where
# the chain is *-Vmiz → *-Vmz → *-Vm (Polomé 1967 p. 808;
# Fulk §5.1; Stentoften runic borumʀ). The *z drops separately by
# EAFFinalZDeletion (later in the pipeline). See DEV_NOTES §17.41
# and dossier-datpl-route-xy-deepdive-2026.md.
define PGmcStarStressedVowel [
    {*á} | {*é} | {*í} | {*ó} | {*ú} |
    # Long vowels in root position (stressed by default in PGmc roots).
    # Needed for forms where PWGmcAiMonophthongization has produced
    # {*ā} (e.g. *stáinamiz → *stānumiz), and similarly for inherited
    # long-V roots that later appear in 3-syllable forms with R2.
    # Long vowels CAN be unstressed in PGmc (e.g. *fedwōr's final *ō,
    # *xáubudą's medial *u), but in those cases they appear in the
    # RIGHT context of R2's pattern, not the left. See DEV_NOTES
    # §17.41 implementation log task 7.
    {*ā} | {*ē} | {*ḗ} | {*ī} | {*ō} | {*ū} |
    # Stressed diphthongs (the *ái diphthong is monophthongized by
    # PWGmcAiMonophthongization before R2 fires, so it is not
    # included here; *áu, *éu, *íu may still be present).
    {*áu} | {*éu} | {*íu}
];
define PWGmcEarlyIApocope Ctx([
    [
    {*i} -> 0 || PGmcStarStressedVowel PGmcStarConsonant+ PGmcStarVocalic PGmcStarConsonant+ _ .#.,
    {*i} -> 0 || PGmcStarStressedVowel PGmcStarConsonant+ PGmcStarVocalic PGmcStarConsonant+ _ {*z} .#.
]
]);

# SC028 Northern West Germanic *xs-cluster simplification (adjudicated 2026:
# sc028-xs-cluster-simplification-adjudication.md).
# *x -> 0 in the cluster *xs + consonant, i.e. only when *s is itself followed
# by a further consonant. This is exactly the handbooks' statement:
#   Campbell §417 p.170: "When a consonant follows, xs > s in OE", with wæstm,
#     -wæsma, North. sesta, and the comparative set ON ísl, ON nýsa, OS weslon,
#     OS wastum, OS niustan, OHG niusen, OS/OHG lastar (< *laxstra-).
#   Brunner §221.2 p.184: "Wenn auf hs andere Konsonanten (auch j) folgen, ist
#     h ausgefallen" — so the following consonant may be *t, *m, *l or *j.
#   Bülbring §527 p.215: "Im Urengl. ist xs vor Konsonanten in der Regel zu s
#     geworden."
#   R/T pp.157-158: "*h was ... lost, possibly variably, possibly only in some

# A-restoration: */æ/ > */a/ before a back vowel (or AFB-fronted *a) in the following syllable.
# Per Ringe/Taylor Vol 2 §6.3.1: "Those stressed *æ which were immediately followed
# by a single or geminate consonant or sC-cluster which was in turn followed by a
# back vowel became a." Ringe explicitly notes: "That unstressed *æ was not retracted."
#
# CHRONOLOGY: A-restoration applies AFTER AFB (Anglo-Frisian Brightening). Ringe says:
# "retraction must have occurred subsequently to fronting—and subsequently to breaking".
# This means at the time A-restoration applies, underlying PWGmc *a has already been
# fronted to surface *æ by AFB.
#
# RESTORATION TRIGGERS: The rule triggers on genuine back vowels in the following
# syllable. After AFB, the remaining back vowels are {*o}, {*u}, {*ō}, {*ū}, etc.
# Suffix *a (like gen.sg. *-as) has been fronted to {*æ} by AFB and does NOT
# trigger restoration (R/T paradigm: *dagas → *dæges → OE dæges, not *dages).
# Only genuine back suffixes like nom.pl. *-os trigger restoration (*dagos → dagas).
#
# WEAK-TAIL EXCLUSION: Only FULL (stressed) vowels trigger restoration. Weak-tail
# unstressed back vowels ({*ą} and the participle/infinitive *-a in inflectional
# position) do NOT trigger restoration, even though they are underlyingly back.
# (Prior to §17.13 this was carried by the breve {*ă}; now the distinction is
# pattern-based via OEARestorationStrongOTail exclusion subtraction.)
# {*ô} (trimoric *ō) is a back vowel that triggers A-restoration, but is not
# in PGmcStarBackVowel to avoid regressions in general vowel rules.
# {*ǭ} (nasalized *ō, from fem. n-stem *-ōn via PNWGmcNStemNLoss) is phonetically
# back and must trigger A-restoration. Without it, *flaskōn wrongly yields flæsċe
# instead of flasce: AFB fronts *a→*æ, then SkPalatalization palatalizes after *æ.
define OEARestorationTriggerVowel [EnglishStarBackVowel | {*ô} | {*ǭ}];

# NWGmc *i → *e lowering before non-high vowels (Campbell §114, Fulk §4.3).
# Campbell: "i > e before mid and low vowels. In OE this change is shown only by
# the common Gmc. words *nest* nest, and *wer* man... and by *spec* bacon beside *spic*."
# Fulk §4.3: "PGmc. i might be lowered to e in parallel fashion before a mid or low
# vowel in the next syllable. Undeniable examples are OE OHG nest < PIE *nizdos."
# Fulk notes the change is "much less systematic" than u-lowering and subject to
# analogical leveling — "leveling away of e being the commonest result."
# Hogg (1992) p.45: treats this as a PGmc/early WGmc change yielding *nesta-.
# Bülbring §81d: "*nëstoz 'Nest' aus älterem *nistoz" — explicit derivation.
# Campbell §116: Redistribution blocked before nasal + consonant (same as u-lowering).
#
# CONSONANT CONDITIONING (Howell & Salmons 1997, Cercignani 1980):
# - Velars and labials BLOCK i-lowering (they share place features with high vowels)
# - Coronals are TRANSPARENT to i-lowering
# - This explains why *nest, *wer show lowering (coronal clusters)
#   but *fish, *lick, *liver retain *i (velar/labial in coda)
#
# ONSET-VELAR BLOCKING (extending Cercignani 1980 to OE):
# Cercignani notes that in Old Icelandic, *i was retained AFTER *k and *g.
# Our OE data suggests this also applies to OE: *xlidą → hlid (not *hled).
# We implement this by requiring no velar ANYWHERE from word-start to *i.
#
# The rule: *i → *e before a non-high vowel, BUT ONLY IF:
# 1. No velar appears anywhere before *i (onset blocking)
# 2. All intervening consonants after *i are coronals (coda blocking)
# 3. *i is in first syllable (stressed) — no vowel appears before *i

# SC023. General (pre-)Proto-Germanic loss of word-final *-n with
# nasalization of the preceding vowel in polysyllables (Ringe 2017

# verbs, and pronouns alike, so this is general phonology, not n-stem

# is Common Germanic at the latest; R/T 2014 pp.54-55 keep the nasalized
# reflex contrastive into PWGmc. The {*ō}{*n}-only environment below is a
# deliberate citation-form proxy: corpus protoforms already encode the
# law's other outputs (*dōną, *wulfą, *sunų), and PGmc *sebun/*nigun/
# *tehun (numeral analogy, Ringe 2017 p.103) plus *hebun- retained -un
# and must not be touched. Live firings are exactly the 17 weak-noun
# citation stems in *-ōn- (stem-form *túngōn → PGmc nom.sg. *túngǭ).
# The *-ǭ is bimoric; it shortens to *-æ → OE -e (unstressed fronting).
# Masc. n-stem nom.sg. uses trimoric *-ô (no -n), which is NOT affected.
# OE dōn is the counterfeeding witness: its secondary final -n (created
# by the later *ą-apocope) survives because this loss was long complete.

# stage claim. See docs/sound_changes/audits/sc023-adjudication.md.
define PNWGmcNStemNLoss [
    {*ō} {*n} -> {*ǭ} || _ .#.
];

define NWGmcInStemNLoss [{*n} -> 0 || {*ī} _ .#.];

# Medial unstressed *u → *o in OE (WS/Northumbrian).
# This is DISTINCT from PNWGmcULowering (stressed u-lowering).
# The exact conditioning is under investigation — see DEV_NOTES.md §14.8.1.
# Campbell §373-374 shows complex conditioning:
# - "protected u > o very often" (e.g. heafod < *xaubud-)
# - BUT "u is always well preserved... in -up, -ung suffixes" (e.g. duguþ, geoguþ)
# The -uþ suffix (< *-unþi-) is a specific exception.
# For now: keep the rule broad, but handle suffix exceptions via lexicon.
# ===========================================================================
# MEDIAL UNSTRESSED U LOWERING (Campbell §373-374)
# ===========================================================================
# Campbell §373: "protected u > o very often" — medial unstressed u tends to
# lower to o in most environments.
#
# HOWEVER, Campbell §373 also states: "u is always well preserved after
# accented u, e.g. sunu, wudu, duguþ" — this is VOWEL HARMONY.
#
# PHONETIC EXPLANATION:
# When the stressed syllable contains [u], the articulatory target for
# backness/roundness is maintained through the word, preventing the lowering
# to [o] that otherwise occurs in unstressed medial position. This is a type
# of regressive vowel harmony (or dissimilation avoidance).
#
# EXAMPLES:
#   *xaubudą → hēafod (first syllable has *au, not *u, so medial u lowers)
#   *jugunθ  → ġeoguþ (first syllable has stressed *u, second u preserved)
#   *dugunθ  → duguþ  (first syllable has stressed *u, second u preserved)
#   *sunuz   → sunu   (both u's preserved)
#   *wuduz   → wudu   (both u's preserved)
#
# IMPLEMENTATION:
# The rule applies u → o in medial position, BUT uses a negative context
# to exclude application when preceded by a syllable containing *u/*ū/*ú.
# When the stressed syllable contains *u (including stressed *ú), the rule
# does not apply, preserving the unstressed *u.
#
# CHRONOLOGY FIX (2026-04-14): This rule MUST run BEFORE PNWGmcULowering
# (stressed *ú → *ó) so that the vowel harmony check can see the stressed *ú.
# If run later, the stressed *ú will have become *ó then *eo, and the harmony
# NOTE ON CHRONOLOGY (see DEV_NOTES.md §14.13):
# Campbell §332: Ingvaeonic nasal loss before voiceless spirants with
# compensatory lengthening: *-unθ- → *-ūθ- (e.g., *dugunθ- → *dugūθ- → duguþ).
# This lengthening MUST happen BEFORE u-lowering, so that long *ū is protected.
# Short *u (as in past plural *-un → -on) undergoes regular lowering.
#
# The harmony check in Campbell §373 ("u preserved after accented u") was 
# misunderstood: duguþ preserves u NOT because of harmony with stressed u,
# but because compensatory lengthening made the medial vowel LONG *ū.
#
# Right-context excludes {*m}: *u is preserved before /m/ per consensus
# (Campbell §373, Brunner §44 Anm. 7, Hogg §3.3.1.3). The protection is
# segmental, not morphological — Campbell §378 shows that when late-WS
# *m > *n the *u immediately lowers. This is what allows DatPl *-um to
# surface as -um (e.g. *skúldrum → sċuldrum). Brunner additionally
# excludes *ng (cf. our parallel OEMedUnstressedILowering for *e→*i / _ *ng);
# not implemented here pending need. See DEV_NOTES §17.41 and
# dossier-medial-u-lowering-conditioning-2026.md.
#
define OEMedUnstressedULowering [
    {*u} -> {*o} || [EnglishStarVocalic - [{*u}|{*ū}|{*ú}]] [EnglishStarConsonant | EnglishPalatalConsonant]+ _ [[EnglishStarConsonant | EnglishPalatalConsonant] - {*m}]
];

# ===========================================================================
# UNSTRESSED *i LOWERING (§17.36)
# ===========================================================================
# Hogg (1992) p.120: "by about 700 all unstressed front vowels had become /e/"
# Campbell §369: "æ, e, and i fell together in a sound written e in unaccented syllables"
# R/T §6.9.6: "The same merger occurred in various derivational suffixes; thus
#   ærist 'first' > ærest, dryhtin 'lord' > dryhten"
#
# We lower plain *i in non-initial syllables to *e. The TSV proto already
# encodes stress in the vowel symbol (acute/macron = stressed; plain = unstressed),
# so a context restricted to "after V+C+" is enough — no auxiliary marking needed.
#
# History (see DEV_NOTES §17.36): an earlier design used an intermediate *ĭ symbol
# produced by OEUnstressedIMarking{1,2,3} and consumed by OEMedUnstressedILowering.
# That auxiliary tier was dismantled in §17.36.1–§17.36.6:
#   §17.36.3: dropped IMarking2 (bi-/ni- prefix marking; OEPrefixIReduction
#             handles the prefix vowel directly).
#   §17.36.4: dropped vacuous bi-/ni- subrule of IMarking3.
#   §17.36.5: folded IMarking1 into a direct *i → *e rewrite.
#   §17.36.6: removed dead IMarking2/3/wrapper definitions and *ĭ clauses
#             from OEMedUnstressedILowering, renamed surviving rule to
#             OEMedUnstressedILowering1.
# ===========================================================================

define OEMedUnstressedILowering1 [
    {*i} -> {*e} || EnglishStarVocalic [EnglishStarConsonant | EnglishPalatalConsonant]+ _
];

# Restore *e → *i before *ng: derivational suffixes *-ing-, *-ung-
# preserve their unstressed vowel uniformly across OE (Campbell §380,
# R/T vol.2 §6.9.6, Hogg 1992 p.120). Implemented as a phonetic context
# restoration since FST rules cannot see morphological boundaries; the
# *ng cluster is the diagnostic.
# Examples: *kuningaz → cyning (not †cyneng), *skíllingaz → sċilling,
# *wīkingaz → wīcing.
# (The *-ig, *-isc, *-iht suffixes mentioned in older comments are
# protected by EnglishPalatalConsonant context handling elsewhere.)
# See DEV_NOTES §17.35 and Germanic/docs/dossier-ing-lowering-2026.md.
define OEMedUnstressedILowering [
    {*e} -> {*i} || _ {*n} {*g}
];

# R/T vol.2 p.303: "So also bi- > be-, ni 'not' > ne."
# Unstressed prefix *i > *ĕ. Use *ĕ (e-breve) for reduced prefix vowels to mark them
# as unstressed and exclude them from subsequent stressed-vowel changes (e.g., palatal
# diphthongization). The *ĕ will later merge with surface e in orthography.
# Restrict to *b and *n to avoid overapplication to root-initial CiC-.
# Further restrict: require the consonant after *i to be followed by a vowel,
# to distinguish prefix bi- (bi-ginn-) from root bind- (b-ind-).
define OEPrefixIReduction [
    {*i} -> {*ĕ} || .#. [{*b} | {*n}] _ [EnglishStarConsonant | EnglishPalatalConsonant] EnglishStarVocalic
];

# ga- prefix: *ga- → *gĕ- (R/T vol.2 §6.9.2-3: unstressed *a in prefixes → *e)
# PGmc *ga- is the perfective/collective prefix (cf. Goth. ga-, OHG gi-/ge-, OE ge-).
# The change *a → *e in unstressed syllables is part of unstressed vowel reduction.
# Use *ĕ (e-breve) to mark the reduced vowel as unstressed, preventing it from
# undergoing palatal diphthongization (*e → *ie after palatals in stressed syllables).
# Restrict to word-initial *g *a followed by consonant + vowel to ensure prefix structure.
define OEPrefixAReduction [
    {*a} -> {*ĕ} || .#. {*g} _ [EnglishStarConsonant | EnglishPalatalConsonant] EnglishStarVocalic
];

# Split into separate composed rules to avoid foma parallel rule context leakage.
# When using [R1, R2] syntax, contextual restrictions can affect all rules.
# Hogg (1992) p.120: "By First Fronting /a/ became /æ/ as in stressed syllables"
# Unstressed suffix *a → *æ (parallel to stressed AFB, but applies before nasals too).
# This feeds OEWeakTailReduction3 (*æ → *e), giving the chain: *a → *æ → *e.
#
# §17.13: Previously the grammar distinguished breve-marked weak *ă from plain
# *a via OEWeakTailReduction1a ({*ă}→{*a}). The OEUnstressedAFronting rule
# ran BEFORE Reduction1a so only genuine unstressed *a (e.g. in *-anaz
# participle suffix) fronted, while transient *ă (e.g. in -janą infinitive
# before OESecondaryNasalization) was protected. With breve eliminated the
# stressed/unstressed partition is now carried entirely by:
#   (a) OESecondaryNasalization protecting coda-nasal *a via *a→*ą before this
#       rule fires, and
#   (b) the right-context requirement below restricting fronting to
#       non-initial medial syllables.
#
# This is NOT a separate sound change but an extension of AFB to unstressed
# positions. AFB (line 1248) explicitly blocks before nasals, which is
# correct for STRESSED syllables (giving OE mann not *mænn). But in
# UNSTRESSED syllables, fronting did occur, as shown by participle
# *-anaz → *-æn → -en.
#
# Example: PGmc *funðanaz → *funðænaz → *funden (via *æ→*e)
# Counter-example: *bakaną → *bakąną (nasalized, protected) → bacan

# SECONDARY NASALIZATION (R/T vol.2 §5.1.2, p.153):
# "Unstressed *a was nasalized, and therefore not fronted, only if it was
# followed by a nasal in the syllable coda."
#
# FIXED 2026-03-13: The original rule context was `_ {*n} .#.` which matched
# both infinitives AND participles AFTER apocope had deleted final vowels.
# This incorrectly nasalized participle -an- suffix, blocking fronting.
#
# The corrected rule runs BEFORE apocope, when final vowels are still present:
#   - Infinitive *bakaną: matches `*a*n*ą.#.` → nasalize → bacan ✓
#   - Participle *bakanăz: has `*a*n*ă*z.#.` → no match → fronts to -en ✓
#
# Examples:
#   Infinitive *bakaną: *bak-a-ną → *bak-ą-ną (nasalized) → bacan
#   Participle *fundanăz: *fund-an-ăz → no match → *fund-æn-ăz → funden
define OESecondaryNasalization [
    {*a} -> {*ą} || _ {*n} .#.
];

define OEUnstressedAFronting [
    # Campbell §333: unaccented *a > *æ (later *e, §369) EXCEPT before a
    # tautosyllabic (coda) nasal. Campbell §334: before a heterosyllabic
    # nasal, fronting DOES apply, yielding e.g. strong past participles
    # in -en < *-anaz.
    # See DEV_NOTES §17.8: the §333 coda-nasal exception is NOT enforced
    # here but by OESecondaryNasalization + OEFinalSchwaApocope chronology:
    # infinitive *-aną → *-ąn (nasalised, protected from fronting);
    # participle *-anăz → *-an (plain *a, fronted to *æ → -en).
    # Any nasal-context guard on this rule over-blocks the participle.
    {*a} -> {*æ} || EnglishStarVocalic [EnglishStarConsonant | EnglishPalatalConsonant]+ _ [EnglishStarConsonant | EnglishPalatalConsonant]
];

# Secondary nasalized vowels (before coda nasal) denasalize to *a, not *ɔ̆.
# Primary nasalized vowels (from lost nasal+fricative) become *ɔ̆.
# R/T vol.2 §5.1.2: unstressed *a before coda nasal was nasalized, blocking fronting.
# After blocking fronting, the nasalization is lost → plain *a.
#
# §17.13: Breve {*ă} eliminated from grammar. OEWeakTailReduction1a
# ({*ă}→{*a}) was the breve-to-plain-a conversion; now vacuous and
# deleted. Nasalization branches (1b, 1c) retained.
define OEWeakTailReduction1b [{*ą} -> {*a} || _ [{*n} | {*m}]];
define OEWeakTailReduction1c [{*ą} -> {*ɔ̆}];
define OEWeakTailReduction1 OEWeakTailReduction1b .o. OEWeakTailReduction1c;

define OEWeakTailReduction3 [

# R/T §6.9.6 (printed pp.334-335): Late unstressed *-ag(-) → -ig(-).
# Chain: medial *a → *e (general unstressed fronting) → *g palatalizes after *e
# → *e raises to *i next to palatal. R/T: "inherited *a adjacent to palatals
# eventually became i. Most strikingly, the inherited derivational suffix
# *-ag(-) > *-æg(-) > *-eg(-) > -ig(-)." Late (after i-umlaut, epenthesis).
# Examples: PGmc *modagaz > OE modig, *hailagaz > halig, *hunagą > hunig.
# Before back vowels the intermediate -eg- can be preserved with velar g
# (e.g. monegum), so palatalization is restricted to word-final position.
define OELateUnstressedAgSuffix (
    # Step 1: front medial unstressed *a → *e before *g
    [{*a} -> {*e} || EnglishStarVocalic [EnglishStarConsonant | EnglishPalatalConsonant]+ _ {*g}]
    .o.
    # Step 2: palatalize *g after *e at word boundary (not before back V; cf. monegum)
    [{*g} -> {*ʝ} || EnglishStarVocalic [EnglishStarConsonant | EnglishPalatalConsonant]+ {*e} _ .#.]
    .o.
    # Late raising is distinct from prehistoric i-mutation.
    [{*e} -> {*i} || EnglishStarVocalic [EnglishStarConsonant | EnglishPalatalConsonant]+ _ [{*ʝ}|{*ʤ}]]
);

# Sievers' Law Syncope (PWGmc): *-CijV- → *-CjV-
# R/T vol.2 p.157: "the sequence *-CijV- was syncopated to *-CjV-"
# This syncope removes the Sievers' Law *i that blocked j-gemination in heavy stems.
#
# Chronology (R/T vol.2 pp.156-157):
# 1. PGmc: Sievers' Law produces *-ija- after heavy stems, *-ja- after light
# 2. PWGmc: j-gemination applies to *Cj → *CCj (light stems only — heavy stems
#    have *Cij, so the C is not directly before j, blocking gemination)
# 3. PWGmc: *-CijV- syncope removes the blocking *i (after gemination!)
# 4. Later: palatalization, etc.
#
# Evidence this is a regular sound change (not analogical leveling):
# - R/T explicitly calls it "syncope" and "syncopated" (p.157)
# - Applies exceptionlessly to "every class I weak present with a heavy root
#   syllable and all ja-stem, jan-stem, and jōn-stem nominals with heavy root
#   syllables" (p.157)
#
# Example derivation:
# PGmc *sōkijană → j-gemination BLOCKED (no *Cj) → syncope *sōkjană → OE sēċan
#
# Note: This rule must be ordered AFTER PWGmcJGemination (so heavy stems avoid
# gemination) and BEFORE palatalization (so *kj → *ċ can apply).
define SieversLawSyncope [
    {*i} -> 0 || [EnglishStarConsonant | EnglishPalatalConsonant] _ {*j}
];

# OE j-retention (orthographic ġ) after *aw-derived long front diphthongs.
# See DEV_NOTES §17.10.36-q3 for full multi-source research synthesis.
#
# R/T vol.1 lines 7169-7172: "*j was not lost after *w" — PGmc-level
# retention of *j in *awj/*iwj environments.
# Kaluza §89(b): "Urg. j im In- und Auslaut nach langen Vokalen = ae. g"
# (strēgan, cīegan, hīeġ are listed in this class).

# Brunner §129 Anm.2: *strawjan > strēgan explicitly derived via diphthong
#   formation + j-retention + i-umlaut (contra short-stem *satjan → settan).
# Lück §98: explicit WS reconstruction "*strīeġan", with critical contrast
#   niwja → niowe (NO -g-) vs strawjan → strēgan (WITH -g-). The rule is
#   NARROWER than "any long diphthong" — *iu-series does not trigger.

# *io (*iu-series) per Lück niwe/niowe contrast.

# Ordering: must fire AFTER OEIUmlaut (so *ēa → *īe umlaut completes)
# and BEFORE OEIntervocalicJVocalization (so *j is not converted to *i).
define OEJStrengtheningAfterFrontDiphthong [

# §17.51.A1: OE combinative u-umlaut on *í (the "w-effect").
# Sources: Bülbring §264; Campbell §218; Brunner §114b; Luick §221;
# Hogg vol.1; R&T §6.9.6; Fulk Comparative Grammar §4.8 (only handbook
# to name the rule explicitly and date it ca. 700).
# Word-initial *w + stressed *í + single C + *u/*o in next syllable
# → *w + *ú + … . Models the OE *wi → *wiu → *wu development
# (Bülbring's two-step chronology) as a single direct change.
# Conditioning notes (per dossier Appendix B):
#   * No consonant restriction in WS (nasals/liquids/velars all permitted).
#   * Trigger restricted to *u/*o (excluding *a, which would over-fire on
#     *wíndaną; combinative *a-umlaut is dialectally restricted per
#     Brunner §114c).
#   * Left-context bare *w- only here; future *sw-/*kw-/*tw- cogset
#     entries would need ({*s}|{*k}|{*t}) {*w} bracket extension.
# Ordering: fires before OEMedUnstressedULowering so the medial *u
# remains visible in the right context, and so that u-lowering's
# stressed-*u left-context exclusion preserves the medial *u under
# the now-stressed *ú.
define OEWICombinativeUUmlaut [
    {*í} -> {*ú} || .#. {*w} _ EnglishStarConsonant [{*u} | {*o}]
];

# R/T vol.2 §6.9.7: WS palatal umlaut.

define OEHCluster [{*x} [{*t} | {*p} | {*s}]];

# SC103 Proto-Germanic nasal loss before *x
# (Campbell §119, p. 44; Fulk §4.1; Ringe 2006 [vol. 1]: "*VN was realized as a
# long nasalized vowel immediately before *h")
# PGmc *-Nx- > long nasalized vowel + *x: the nasal is lost with compensatory
# lengthening and nasalization of the preceding vowel. This is PAN-GERMANIC,
# shared by Gothic, Norse and the whole of West Germanic — Go. þeihan, brāhta,
# þūhta; OHG dīhan, brāhta, dūhta, fūht, fūst; OE þēon, brōhte, þūhte, fūht.
# It is NOT the later North Sea Germanic nasal-spirant law below, every one of
# whose witnesses keeps the nasal outside the Ingvaeonic area (OHG fimf, gans,
# ander, jugund). Campbell §121 and Fulk §4.11 both present the Ingvaeonic law
# as a separate, later change merely "similar"/"comparable" to this one.
# Only *a, *i, *u occur here: Fulk §4.1 "The vowels e and o did not occur in
# this environment"; cf. Campbell §116 n. 4, §119.
# WHAT THE RULE DOES: despite the short identifier, this is the full
# three-part change of Campbell §119 p. 44 — "loss of the nasal consonant, and
# compensatory lengthening and nasalization of the vowel" (Fulk's heading for
# §4.1 p. 55 is "Compensatory lengthening upon loss of a nasal consonant").
# A SHORT input vowel therefore yields a LONG one; the acute-marked inputs lose
# their stress diacritic because CAPR has no stressed long low/back tier, so
# stress is absorbed into length exactly as in EAFNasalSpirantLengthening.
# OUTPUT IS NASALIZED, NOT ROUNDED. Fulk §4.1 p. 55: the lengthened vowels
# "may have remained nasalized for a considerable time, well past the close of
# the NWGmc. period, since ą̄ produced this way developed to ō in Anglo-Frisian
# ... and did not fall together with OE ā < ai". Ringe vol. 1 §3.2.7(ii)
# pp. 149-50 reconstructs PGmc *hanhaną = *[xɑ̃:xanɑ̃] outright. The rounding is
# a much later Anglo-Frisian development shared with the two other sources of
# *ą̄ (EAFNasalizedLowRounding below); a Proto-Germanic rule must not
# anticipate it, or the Gothic/Norse/OHG/OS outcome would be misrepresented.
# Campbell §119 p. 44 is explicit that this *ą̄ "became ā in Goth., North Gmc.,
# OHG, and OS", and only "in OE and OFris." was it rounded (Go. OHG brāhta :
# OE brōhte, Fulk §4.11 p. 72).
# The high vowels take NO explicit nasalized tier: Campbell §119 p. 44 and §121
# p. 47 state that *ī̃ and *ū̃ "subsequently develop like original ī and ū", and
# Brunner §67 p. 59 says nasalized *ī merged with old *ī "nach Aufgabe der
# Nasalierung". Their nasalization has no independent consequence, so
# representing it would add unused states. See the adjudication memo.
define PGmcNasalLossBeforeX Ctx([
    [
    {*a} -> {*ã} || _ EnglishStarNasal {*x},
    {*i} -> {*ī} || _ EnglishStarNasal {*x},
    {*u} -> {*ū} || _ EnglishStarNasal {*x},
    {*á} -> {*ã} || _ EnglishStarNasal {*x},
    {*í} -> {*ī} || _ EnglishStarNasal {*x},
    {*ú} -> {*ū} || _ EnglishStarNasal {*x}
] .o. [
    EnglishStarNasal -> 0 || _ {*x}
]
]);

# SC026 North Sea Germanic nasal-spirant law, step 1 of 2:
# nasalization and compensatory lengthening of the vowel.
# R/T vol. 2 §5.1.1 (pp. 139-41): "The most obvious phonological innovation of
# the northern dialects is the loss of nasals immediately preceding fricatives,
# with lengthening and nasalization of the preceding vowel."
# HISTORICALLY THIS AND EAFNasalSpirantLoss ARE ONE SOUND CHANGE. Every handbook
# states the loss and the compensatory lengthening as a single development
# (Campbell §121; Fulk §4.11; Sievers-Brunner §186.1; Luick §301.1). CAPR splits
# it into two transducer operations only because the vowel must be read while
# the conditioning nasal is still present, before the nasal is deleted. The
# order EAFNasalSpirantLengthening < EAFNasalSpirantLoss is therefore an
# EXECUTABLE DEPENDENCY, not a demonstrated historical relative chronology.
# Domain: *mf, *nþ, *ns only (EnglishStarNSGmcSpirant), never *nx.
# Input vowels: *a, *i, *u only — *e and *o do not occur before nasal + obstruent
# in Germanic (Fulk §4.1), and *æ arises far later than this stage.
# OUTPUT FOR *a IS *ą̄, NOT *ō. Campbell §121 p. 47: the groups "reject the
# nasal consonant with compensatory lengthening and nasalization of the
# preceding vowel to ą̄, ī̃, ū̃ ... West Gmc. ą̄ > ā in OS only, while in OFris.
# and OE it remained nasalized, and was ultimately developed to ō". Fulk §4.11
# p. 72: "The change thus affects mf, ns, nþ and produces ą̄, ī̃, ū̃. The first of
# these yields ō in Anglo-Frisian (as does ą̄ inherited from PGmc.)". R/T vol. 2
# §5.1.1 pp. 139-41 write every outcome as a nasalized vowel (*gą̄s, *ą̄sti,
# *ą̄sla). The rounding is the separate, later EAFNasalizedLowRounding.
# The high branches stay unnasalized in the model: Campbell §121 p. 47 says
# *ī̃/*ū̃ "subsequently develop like original ī and ū"; nothing later depends on
# their nasality (Brunner §67 p. 59, §186.1).
# CHRONOLOGY: must precede OE i-umlaut, or umlauted vowels are not lengthened.
define EAFNasalSpirantLengthening [
    {*a} -> {*ã} || _ EnglishStarNasal EnglishStarNSGmcSpirant,
    {*i} -> {*ī} || _ EnglishStarNasal EnglishStarNSGmcSpirant,
    {*u} -> {*ū} || _ EnglishStarNasal EnglishStarNSGmcSpirant,
    # Stressed variants (*á -> *ã: stress absorbed into length, no stressed
    # long low tier exists):
    {*á} -> {*ã} || _ EnglishStarNasal EnglishStarNSGmcSpirant,
    {*í} -> {*ī} || _ EnglishStarNasal EnglishStarNSGmcSpirant,
    {*ú} -> {*ū} || _ EnglishStarNasal EnglishStarNSGmcSpirant
];

# SC027 North Sea Germanic nasal-spirant law, step 2 of 2: deletion of the

# the second executable operation of ONE historical sound change.
define EAFNasalSpirantLoss [
    EnglishStarNasal -> 0 || _ EnglishStarNSGmcSpirant
];

# Pragmatic approach: encode heavy-syllable apocope directly rather than using
# a two-step mark-then-delete system (which misbehaves under composition due to
# foma's character-level symbol handling).
# R/T §6.8.1: "short *i and *u were lost word-finally after a heavy syllable."
# Heavy syllable = long vowel/diphthong + any consonants, OR short vowel + 2+ consonants.
# OEAnyConsonant broadens the consonant class to include palatalized consonants
# produced by earlier OE rules (e.g. *ʤ from palatalization).
define OEAnyConsonant [EnglishStarConsonant | EnglishPalatalConsonant];
define OEHeavyMarker ?*;  # identity — no marking needed
define OEWeightMarkers OEHeavyMarker;

define OEHighVowelApocope [
    # Disyllabic: long syllable (long vowel + consonant(s)) + final vowel
    {*i} -> 0 || EnglishStarLongVowel OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarLongVowel OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarLongVowel OEAnyConsonant+ _ .#.,
    # Disyllabic: diphthong + consonant(s) + final vowel
    # LONG diphthongs are heavy (any C count). SHORT diphthongs are heavy only
    # with 2+ consonants; a short diphthong + single C is a LIGHT syllable (from
    # back-umlaut of a light stem, Campbell §345; DEV_NOTES §17.17).
    # E.g. *spéru → speoru (LIGHT, retain -u); *xérdō → heord (HEAVY, lose -u).
    {*i} -> 0 || EnglishStarLongDiphthong OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarLongDiphthong OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarLongDiphthong OEAnyConsonant+ _ .#.,
    {*i} -> 0 || EnglishStarShortDiphthong OEAnyConsonant OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarShortDiphthong OEAnyConsonant OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarShortDiphthong OEAnyConsonant OEAnyConsonant+ _ .#.,
    # Disyllabic: heavy closed syllable (short vowel + 2+ consonants) + final vowel
    {*i} -> 0 || EnglishStarShortVowel OEAnyConsonant OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarShortVowel OEAnyConsonant OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarShortVowel OEAnyConsonant OEAnyConsonant+ _ .#.,
    # Trisyllabic: long/diphthong + unstressed syllable + final vowel (Campbell §345)
    # Pattern: LongV/Diph + C+ + shortV + C+ + final_vowel
    # Example: heofon < *xemonų: *e*o + *β + *u + *n + *ų
    {*i} -> 0 || EnglishStarLongVowel OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarLongVowel OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarLongVowel OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*i} -> 0 || EnglishStarLongDiphthong OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarLongDiphthong OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarLongDiphthong OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*i} -> 0 || EnglishStarShortDiphthong OEAnyConsonant OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarShortDiphthong OEAnyConsonant OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarShortDiphthong OEAnyConsonant OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    # Trisyllabic with short-diphthong first syllable (from back-umlaut of a light
    # stem): trisyllabic apocope still fires regardless of stress-syllable weight
    # (Campbell §345). E.g. *xémonų → *xéomonų → heofon.
    {*i} -> 0 || EnglishStarShortDiphthong OEAnyConsonant EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarShortDiphthong OEAnyConsonant EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarShortDiphthong OEAnyConsonant EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    # Trisyllabic: short syllable + another syllable + final vowel (Campbell §345)
    # Pattern: shortV + C + shortV + C+ + final_vowel
    {*i} -> 0 || EnglishStarShortVowel OEAnyConsonant EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarShortVowel OEAnyConsonant EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarShortVowel OEAnyConsonant EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    # Trisyllabic with heavy-by-position first syllable (short V + 2+C) +
    # light second syllable + final high V (Campbell §345; DEV_NOTES §17.45.4).
    # E.g. *spénnilō → *spinnilu → *spinnil (then OELAdjacentSyncope + nn-degem → spinl).
    {*i} -> 0 || EnglishStarShortVowel OEAnyConsonant OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*u} -> 0 || EnglishStarShortVowel OEAnyConsonant OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    {*ų} -> 0 || EnglishStarShortVowel OEAnyConsonant OEAnyConsonant+ EnglishStarShortVowel OEAnyConsonant+ _ .#.,
    # R/T §6.6.1: vowel-hiatus contraction — word-final *i after long vowel
    # with no intervening consonant (e.g. root noun dat.sg. *kūi → cȳ)
    {*i} -> 0 || EnglishStarLongVowel _ .#.,
    # Campbell §238/§346: final -u lost after /h/ (*x) regardless of weight.
    # Must fire before intervocalic h-loss would remove the conditioning.
    # E.g. *féxu → *féoxu → feoh (not feou).
    {*u} -> 0 || {*x} _ .#.,
    {*ų} -> 0 || {*x} _ .#.,
    {*i} -> 0 || {*x} _ .#.
] .o. [{*ᶜ} -> 0];

define OELAdjacentSyncope [

    {*i} -> 0 || EnglishStarLongVowel OEAnyConsonant+ _ {*l},

# R-metathesis: Campbell §459, R/T §6.8.2 (p.341)
# OE metathesized r from before to after a short vowel when followed by s+C cluster.

# Examples: *brestăną → berstan, *frustą → forst
# Campbell §459(2): "Metathesis is much less frequent when d followed the vowel"
